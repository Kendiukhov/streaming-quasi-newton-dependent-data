%% Information Sciences (Elsevier) submission.
%%
%% Built with Elsevier's elsarticle class.  Single-column ("preprint") layout, which is what
%% Elsevier asks for at submission; "5p" or "3p" produce the typeset journal layout and are not
%% used here.  Line numbering is on to help reviewers refer to specific passages.
%%
%% This file owns everything venue-specific: title page, abstract, keywords, float placement, the
%% declarations Information Sciences requires, and the appendix ordering.  The scientific prose is
%% shared verbatim with the preprint version in ../paper, so the two cannot drift apart:
%%   ../paper/abstract.tex        the abstract (200 words, the journal's limit)
%%   ../paper/sec_*.tex           the section bodies
%%   numbers.tex         every numeric claim, generated from results/*.json
%%   ../paper/tab_*.tex           every table, generated from results/*.json
%% Four figures that do not fit the journal's 10-item guideline on figures + tables are in
%% supplementary.tex; the macros \RefLrvRate, \RefGapFig, \RefCostFig and \RefAblFig point the
%% shared prose at them.
\documentclass[preprint,11pt,authoryear]{elsarticle}

% \overfullrule=8pt   % uncomment to mark overfull lines with a black bar while editing
% Page measure.  elsarticle's "preprint" mode sets a deliberately narrow 5-inch text block, which
% inflates the page count of a manuscript by roughly half.  Information Sciences imposes no formatting
% requirements at submission, so we use an ordinary one-inch measure -- legible, single column, line
% numbered -- with extra room on the left for the line numbers.
\usepackage[left=1.25in,right=1in,top=1in,bottom=1.5in,footskip=0.75in]{geometry}
\usepackage{lineno}
\linenumbers

% Let a page finish short rather than push a stretched line past the bottom margin.
\raggedbottom

\usepackage{amsmath,amssymb,amsthm,mathtools}
\usepackage[T1]{fontenc}
\usepackage{microtype}
\usepackage{booktabs}
\usepackage{multirow}
\usepackage{graphicx}
% Figure inclusion, shared by both documents.  \IfFileExists does NOT consult \graphicspath, so
% the guard has to test the same path graphicx will use -- getting this wrong silently replaces
% every figure with a placeholder box, which is exactly what happened once.
\newcommand{\FIGDIR}{../figures/}
\graphicspath{{\FIGDIR}}
\newcommand{\figorbox}[1]{%
  \IfFileExists{\FIGDIR#1}{\includegraphics[width=\linewidth]{#1}}%
  {\fbox{\parbox{0.9\linewidth}{\centering\vspace{2em}%
    \texttt{\detokenize{#1}} pending: run \texttt{make figures}\vspace{2em}}}}}

\usepackage{xcolor}
\usepackage[colorlinks=true,linkcolor=blue!55!black,citecolor=blue!55!black,
            urlcolor=blue!55!black]{hyperref}
\usepackage{cleveref}   % before \newtheorem so cleveref can record the environment names
\usepackage{enumitem}

% Float placement.  Ten floats inside one section overflows LaTeX's defaults and they get
% flushed to the end of the document -- which once pushed three figures past the appendices.
% Raise the per-page limits, allow float pages, and put a barrier before the appendix.
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{5}
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.7}
\renewcommand{\textfraction}{0.06}
\renewcommand{\floatpagefraction}{0.72}
\usepackage{placeins}
\usepackage{caption}
% The ablation table has 77 body rows and cannot fit one page.  A float taller than the text block
% is not an error in LaTeX -- it simply runs off the bottom, silently losing rows -- so that table
% is a longtable, which breaks across pages instead.
\usepackage{longtable}
% ... and needspace, because a longtable's first chunk is atomic: caption + head + the first
% break-protected panel is ~308pt with no legal breakpoint inside it, so on a page offering less
% than that it silently overruns the bottom margin -- which it did, by 0.77in on page 44, with no
% warning from LaTeX and none from the geometry gate either.
\usepackage{needspace}
\captionsetup{font=small,labelfont=bf}
